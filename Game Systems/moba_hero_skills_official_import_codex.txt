  PASSIVE: 2n,
  NO_TARGET: 4n,

  OPTIONAL_POINT: 16384n,
  OPTIONAL_NO_TARGET: 32768n,
  AURA: 65536n,
  ATTACK: 131072n,
  DONT_RESUME_MOVEMENT: 262144n,
  ROOT_DISABLES: 524288n,
  UNRESTRICTED: 1048576n,
  IGNORE_PSEUDO_QUEUE: 2097152n,
  IGNORE_CHANNEL: 4194304n,
  DONT_CANCEL_MOVEMENT: 8388608n,
  DONT_ALERT_TARGET: 16777216n,
  DONT_RESUME_ATTACK: 33554432n,
  NORMAL_WHEN_STOLEN: 67108864n,
  IGNORE_BACKSWING: 134217728n,
  RUNE_TARGET: 268435456n,
  DONT_CANCEL_CHANNEL: 536870912n,
  VECTOR_TARGETING: 1073741824n,
};

export function decodeBehaviorFlags(raw: string): string[] {
  const value = BigInt(raw || "0");

  return Object.entries(ABILITY_BEHAVIOR_FLAGS)
    .filter(([, bit]) => (value & bit) !== 0n)
    .map(([name]) => name);
}

type JsonRecord = Record<string, any>;

async function fetchJson<T>(url: string, attempts = 4): Promise<T> {
  let lastError: unknown;

  for (let attempt = 1; attempt <= attempts; attempt += 1) {
    try {
      const response = await fetch(url, {
        headers: {
          "user-agent": "moba-manager-hero-import/1.0",
          accept: "application/json",
        },
      });

      if (!response.ok) {
        throw new Error(`HTTP ${response.status} em ${url}`);
      }

      return (await response.json()) as T;
    } catch (error) {
      lastError = error;

      if (attempt < attempts) {
        await new Promise((done) => setTimeout(done, 500 * attempt));
      }
    }
  }

  throw lastError;
}

async function mapWithConcurrency<T, R>(
  values: T[],
  concurrency: number,
  mapper: (value: T, index: number) => Promise<R>
): Promise<R[]> {
  const output = new Array<R>(values.length);
  let nextIndex = 0;

  async function worker() {
    while (true) {
      const index = nextIndex;
      nextIndex += 1;

      if (index >= values.length) return;

      output[index] = await mapper(values[index], index);
    }
  }

  await Promise.all(
    Array.from({ length: Math.min(concurrency, values.length) }, worker)
  );

  return output;
}

function normalizeNumbers(values: unknown): number[] {
  if (!Array.isArray(values)) return [];

  return values
    .map((value) => Number(value))
    .filter((value) => Number.isFinite(value));
}

function classifyAbility(ability: JsonRecord): SkillCategory {
  if (ability.ability_is_innate) return "innate";
  if (ability.ability_is_granted_by_scepter) return "scepter_granted";
  if (ability.ability_is_granted_by_shard) return "shard_granted";
  if (ability.is_item) return "other";
  return "standard";
}

function genericSlotFor(
  category: SkillCategory,
  categoryIndex: number
): string {
  switch (category) {
    case "standard":
      return `standard_${categoryIndex + 1}`;
    case "innate":
      return `innate_${categoryIndex + 1}`;
    case "scepter_granted":
      return `scepter_granted_${categoryIndex + 1}`;
    case "shard_granted":
      return `shard_granted_${categoryIndex + 1}`;
    default:
      return `other_${categoryIndex + 1}`;
  }
}

function normalizeSpecialValue(value: JsonRecord): OfficialSpecialValue {
  return {
    name: String(value.name ?? ""),
    heading: String(value.heading_loc ?? ""),
    baseValues: normalizeNumbers(value.values_float),
    isPercentage: Boolean(value.is_percentage),
    shardValues: normalizeNumbers(value.values_shard),
    scepterValues: normalizeNumbers(value.values_scepter),
    bonuses: Array.isArray(value.bonuses)
      ? value.bonuses.map((bonus: JsonRecord) => ({
          name: String(bonus.name ?? ""),
          value: Number(bonus.value ?? 0),
          operation: Number(bonus.operation ?? 0),
        }))
      : [],
    facetBonus: {
      name: String(value.facet_bonus?.name ?? ""),
      values: normalizeNumbers(value.facet_bonus?.values),
      operation: Number(value.facet_bonus?.operation ?? 0),
    },
    requiredFacet: String(value.required_facet ?? ""),
  };
}

function normalizeAbility(
  ability: JsonRecord,
  orderInOfficialFeed: number,
  categoryIndex: number
): OfficialAbilitySource {
  const category = classifyAbility(ability);

  return {
    sourceAbilityId: Number(ability.id),
    sourceInternalName: String(ability.name ?? ""),
    sourceDisplayName: String(ability.name_loc ?? ""),
    sourceDescription: String(ability.desc_loc ?? ""),
    sourceLore: String(ability.lore_loc ?? ""),
    sourceNotes: Array.isArray(ability.notes_loc)
      ? ability.notes_loc.map(String)
      : [],

    category,
    orderInOfficialFeed,
    genericSlot: genericSlotFor(category, categoryIndex),

    maxLevel: Number(ability.max_level ?? 0),
    typeRaw: Number(ability.type ?? 0),
    behaviorRaw: String(ability.behavior ?? "0"),
    behaviorFlags: decodeBehaviorFlags(String(ability.behavior ?? "0")),

    targetTeamRaw: Number(ability.target_team ?? 0),
    targetTypeRaw: Number(ability.target_type ?? 0),
    abilityFlagsRaw: Number(ability.flags ?? 0),
    damageTypeRaw: Number(ability.damage ?? 0),
    immunityRaw: Number(ability.immunity ?? 0),
    dispellableRaw: Number(ability.dispellable ?? 0),

    castRanges: normalizeNumbers(ability.cast_ranges),
    castPoints: normalizeNumbers(ability.cast_points),
    channelTimes: normalizeNumbers(ability.channel_times),
    cooldowns: normalizeNumbers(ability.cooldowns),
    durations: normalizeNumbers(ability.durations),
    damages: normalizeNumbers(ability.damages),
    manaCosts: normalizeNumbers(ability.mana_costs),
    healthCosts: normalizeNumbers(ability.health_costs),
    goldCosts: normalizeNumbers(ability.gold_costs),

    specialValues: Array.isArray(ability.special_values)
      ? ability.special_values.map(normalizeSpecialValue)
      : [],

    upgrades: {
      hasScepterUpgrade: Boolean(ability.ability_has_scepter),
      hasShardUpgrade: Boolean(ability.ability_has_shard),
      grantedByScepter: Boolean(ability.ability_is_granted_by_scepter),
      grantedByShard: Boolean(ability.ability_is_granted_by_shard),
      scepterText: String(ability.scepter_loc ?? ""),
      shardText: String(ability.shard_loc ?? ""),
      facetsText: Array.isArray(ability.facets_loc)
        ? ability.facets_loc.map(String)
        : [],
    },

    isInnate: Boolean(ability.ability_is_innate),
  };
}

function groupAbilities(abilities: OfficialAbilitySource[]) {
  return {
    standardAbilities: abilities.filter(
      (ability) => ability.category === "standard"
    ),
    innateAbilities: abilities.filter(
      (ability) => ability.category === "innate"
    ),
    scepterGrantedAbilities: abilities.filter(
      (ability) => ability.category === "scepter_granted"
    ),
    shardGrantedAbilities: abilities.filter(
      (ability) => ability.category === "shard_granted"
    ),
    otherAbilities: abilities.filter(
      (ability) => ability.category === "other"
    ),
  };
}

function toAdaptedSkill(
  legacyHeroId: string,
  sourceHeroId: number,
  ability: OfficialAbilitySource,
  order: number
): AdaptedHeroSkill {
  const categoryLabel = ability.category.replaceAll("_", " ");

  return {
    id: `${legacyHeroId}_${ability.category}_${order + 1}_${ability.sourceAbilityId}`,
    legacyHeroId,
    sourceHeroId,
    sourceAbilityId: ability.sourceAbilityId,

    displayName:
      `${categoryLabel.charAt(0).toUpperCase()}${categoryLabel.slice(1)} ${order + 1}`,
    category: ability.category,
    genericSlot: ability.genericSlot,
    order,

    maxLevel: ability.maxLevel,
    behaviorFlags: ability.behaviorFlags,
    behaviorRaw: ability.behaviorRaw,

    targetTeamRaw: ability.targetTeamRaw,
    targetTypeRaw: ability.targetTypeRaw,
    damageTypeRaw: ability.damageTypeRaw,
    immunityRaw: ability.immunityRaw,
    dispellableRaw: ability.dispellableRaw,

    castRanges: ability.castRanges,
    castPoints: ability.castPoints,
    channelTimes: ability.channelTimes,
    cooldowns: ability.cooldowns,
    durations: ability.durations,
    damages: ability.damages,
    manaCosts: ability.manaCosts,
    healthCosts: ability.healthCosts,
    goldCosts: ability.goldCosts,

    specialValues: ability.specialValues,

    upgrades: {
      hasScepterUpgrade: ability.upgrades.hasScepterUpgrade,
      hasShardUpgrade: ability.upgrades.hasShardUpgrade,
      grantedByScepter: ability.upgrades.grantedByScepter,
      grantedByShard: ability.upgrades.grantedByShard,
      scepterText: ability.upgrades.scepterText,
      shardText: ability.upgrades.shardText,
    },

    sourceMetadata: {
      internalAbilityName: ability.sourceInternalName,
      officialDisplayName: ability.sourceDisplayName,
      officialDescription: ability.sourceDescription,
      notes: ability.sourceNotes,
    },
  };
}

function adaptGroup(
  legacyHeroId: string,
  sourceHeroId: number,
  abilities: OfficialAbilitySource[]
): AdaptedHeroSkill[] {
  return abilities.map((ability, index) =>
    toAdaptedSkill(legacyHeroId, sourceHeroId, ability, index)
  );
}

function serializeTsExport(exportName: string, value: unknown): string {
  return [
    "/* eslint-disable */",
    "// Arquivo gerado automaticamente. Não editar manualmente.",
    `export const ${exportName} = ${JSON.stringify(value, null, 2)} as const;`,
    "",
  ].join("\n");
}

async function ensureWrite(path: string, content: string) {
  await mkdir(dirname(path), { recursive: true });
  await writeFile(path, content, "utf8");
}

async function main() {
  const heroListResponse = await fetchJson<JsonRecord>(HERO_LIST_URL);
  const officialHeroes: JsonRecord[] =
    heroListResponse?.result?.data?.heroes ?? [];

  if (officialHeroes.length === 0) {
    throw new Error("O herolist oficial retornou vazio.");
  }

  const officialById = new Map<number, JsonRecord>(
    officialHeroes.map((hero) => [Number(hero.id), hero])
  );

  const mappedEntries = Object.entries(
    LEGACY_HERO_TO_OFFICIAL_SOURCE
  );

  const mappedSourceIds = mappedEntries.map(
    ([, mapping]) => mapping.sourceHeroId
  );

  if (new Set(mappedSourceIds).size !== mappedSourceIds.length) {
    throw new Error(
      "O mapping possui sourceHeroId duplicado. Corrija antes de importar."
    );
  }

  const unmappedOfficialHeroes = officialHeroes
    .filter((hero) => !mappedSourceIds.includes(Number(hero.id)))
    .map((hero) => ({
      sourceHeroId: Number(hero.id),
      sourceName: String(hero.name_loc ?? hero.name ?? ""),
    }));

  const missingSourceHeroes = mappedEntries
    .filter(([, mapping]) => !officialById.has(mapping.sourceHeroId))
    .map(([legacyHeroId, mapping]) => ({
      legacyHeroId,
      ...mapping,
    }));

  if (missingSourceHeroes.length > 0) {
    throw new Error(
      `Existem mappings ausentes no herolist oficial: ${JSON.stringify(missingSourceHeroes)}`
    );
  }

  await ensureWrite(
    resolve("reports/unmapped-official-heroes.json"),
    JSON.stringify(unmappedOfficialHeroes, null, 2)
  );

  const capturedAtIso = new Date().toISOString();

  const sourceKits = await mapWithConcurrency(
    mappedEntries,
    5,
    async ([legacyHeroId, mapping]) => {
      const response = await fetchJson<JsonRecord>(
        heroDataUrl(mapping.sourceHeroId)
      );

      const hero = response?.result?.data?.heroes?.[0];

      if (!hero) {
        throw new Error(
          `herodata vazio para sourceHeroId ${mapping.sourceHeroId}`
        );
      }

      const officialName = String(hero.name_loc ?? "");

      if (
        mapping.expectedSourceName &&
        officialName !== mapping.expectedSourceName
      ) {
        throw new Error(
          `Mapping divergente para ${legacyHeroId}: esperado "${mapping.expectedSourceName}", recebido "${officialName}".`
        );
      }

      const rawAbilities: JsonRecord[] = Array.isArray(hero.abilities)
        ? hero.abilities
        : [];

      const categoryCounters: Record<SkillCategory, number> = {
        standard: 0,
        innate: 0,
        scepter_granted: 0,
        shard_granted: 0,
        other: 0,
      };

      const abilities = rawAbilities.map((ability, order) => {
        const category = classifyAbility(ability);
        const normalized = normalizeAbility(
          ability,
          order,
          categoryCounters[category]
        );
        categoryCounters[category] += 1;
        return normalized;
      });

      return {
        legacyHeroId,
        sourceHeroId: Number(hero.id),
        sourceInternalName: String(hero.name ?? ""),
        sourceDisplayName: officialName,
        capturedAtIso,
        facets: Array.isArray(hero.facets) ? hero.facets : [],
        ...groupAbilities(abilities),
      } satisfies OfficialHeroSkillSource;
    }
  );

  const adaptedKits: AdaptedHeroSkillKit[] = sourceKits.map((kit) => ({
    heroId: kit.legacyHeroId,
    sourceHeroId: kit.sourceHeroId,
    capturedAtIso: kit.capturedAtIso,
    facets: kit.facets,

    standardAbilities: adaptGroup(
      kit.legacyHeroId,
      kit.sourceHeroId,
      kit.standardAbilities
    ),
    innateAbilities: adaptGroup(
      kit.legacyHeroId,
      kit.sourceHeroId,
      kit.innateAbilities
    ),
    scepterGrantedAbilities: adaptGroup(
      kit.legacyHeroId,
      kit.sourceHeroId,
      kit.scepterGrantedAbilities
    ),
    shardGrantedAbilities: adaptGroup(
      kit.legacyHeroId,
      kit.sourceHeroId,
      kit.shardGrantedAbilities
    ),
    otherAbilities: adaptGroup(
      kit.legacyHeroId,
      kit.sourceHeroId,
      kit.otherAbilities
    ),
  }));

  const report = {
    capturedAtIso,
    officialHeroCount: officialHeroes.length,
    mappedUniqueHeroCount: sourceKits.length,
    deprecatedLegacyHeroes: DEPRECATED_LEGACY_HERO_IDS,
    unmappedOfficialHeroes,
    heroesWithNonFourStandardAbilityCount: sourceKits
      .filter((kit) => kit.standardAbilities.length !== 4)
      .map((kit) => ({
        legacyHeroId: kit.legacyHeroId,
        sourceHeroId: kit.sourceHeroId,
        sourceDisplayName: kit.sourceDisplayName,
        standardAbilityCount: kit.standardAbilities.length,
        innateCount: kit.innateAbilities.length,
        scepterGrantedCount: kit.scepterGrantedAbilities.length,
        shardGrantedCount: kit.shardGrantedAbilities.length,
      })),
  };

  await ensureWrite(
    resolve("src/data/heroSkillSourceOfficial.ts"),
    serializeTsExport("HERO_SKILL_SOURCE_OFFICIAL", sourceKits)
  );

  await ensureWrite(
    resolve("src/data/heroSkillKitsAdapted.ts"),
    serializeTsExport("HERO_SKILL_KITS_ADAPTED", adaptedKits)
  );

  await ensureWrite(
    resolve("src/data/legacyHeroSkillMigration.ts"),
    [
      "/* eslint-disable */",
      "// Arquivo gerado automaticamente.",
      `export const LEGACY_HERO_TO_OFFICIAL_SOURCE = ${JSON.stringify(
        LEGACY_HERO_TO_OFFICIAL_SOURCE,
        null,
        2
      )} as const;`,
      `export const DEPRECATED_LEGACY_HERO_IDS = ${JSON.stringify(
        DEPRECATED_LEGACY_HERO_IDS,
        null,
        2
      )} as const;`,
      "",
    ].join("\n")
  );

  await ensureWrite(
    resolve("reports/hero-skill-import-report.json"),
    JSON.stringify(report, null, 2)
  );

  console.log(
    `Importação concluída: ${sourceKits.length} heróis únicos.`
  );

  if (unmappedOfficialHeroes.length > 0) {
    console.warn(
      "Há heróis oficiais novos sem mapping:",
      unmappedOfficialHeroes
    );
  }
}

main().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
