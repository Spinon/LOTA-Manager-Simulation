// 7. ESCOLHA DE ALVO
// ============================================================================

export interface TargetScoreBreakdown {
  targetId: string;

  strategicValue: number;
  currentThreat: number;
  killProbability: number;
  accessibility: number;
  allyFollowUp: number;
  positioningError: number;
  cooldownPunishValue: number;
  interruptValue: number;
  objectiveConversionValue: number;

  defensiveResources: number;
  enemySaveCoverage: number;
  overextensionRisk: number;
  baitRisk: number;
  expectedOverkill: number;
  targetSwitchCost: number;

  finalScore: number;
  reasons: string[];
}

targetScore =
  strategicValue
  + currentThreat
  + killProbability
  + accessibility
  + allyFollowUp
  + positioningError
  + cooldownPunishValue
  + interruptValue
  + objectiveConversionValue
  - defensiveResources
  - enemySaveCoverage
  - overextensionRisk
  - baitRisk
  - expectedOverkill
  - targetSwitchCost;

// Regras:
//
// - carry não é sempre prioridade;
// - channeling importante pode virar prioridade máxima;
// - alvo já controlado e com dano suficiente recebe penalidade de overkill;
// - alvo sem escape e fora de posição recebe bônus;
// - backline só é alvo válido se acessibilidade for suficiente;
// - suporte com save disponível pode ser prioridade antes do core;
// - iniciador sem recursos pode ser ignorado.

// ============================================================================
// 8. COORDENAÇÃO DE CC
// ============================================================================

followUpCastTime =
  currentControlEnd
  - ownCastPoint
  - projectileTravelTime
  - reactionMargin;

reactionMargin =
  baseReactionMargin
  + communicationDelay
  + playerExecutionDelay;

controlOverlapPenalty =
  overlapDuration
  * controlValuePerSecond
  * coordinationPenaltyMultiplier;

controlChainScore =
  totalControlDuration
  + controlReliability
  + burstFollowUp
  + escapeDenial
  - overlapPenalty
  - dispelRisk
  - immunityRisk;

// Ignorar overlap penalty quando:
//
// - precisa interromper channeling;
// - alvo vai escapar;
// - controle atual pode ser removido;
// - novo controle pega múltiplos alvos;
// - morte imediata abre objetivo;
// - aliado precisa ser salvo agora.
//
// Ordem sugerida:
//
// 1. displacement/setup;
// 2. hard CC;
// 3. follow-up hard CC;
// 4. silence/leash/root;
// 5. slow;
// 6. finisher.

// ============================================================================
// 9. RESERVA DE DANO E ANTI-OVERKILL
// ============================================================================

expectedCommittedDamage =
  sum(
    reservation.expectedDamage
    * reservation.reliability
  );

remainingRequiredDamage =
  targetEffectiveHealth
  - expectedCommittedDamage;

overkillPenalty =
  max(0, expectedCommittedDamage - targetEffectiveHealth)
  * ultimateCostWeight;

// Exceções:
//
// - target possui death prevention;
// - target tem heal/sustain alto;
// - target recebe save externo;
// - dano reservado é pouco confiável;
// - morte instantânea abre objetivo decisivo.

// ============================================================================
// 10. SAVES E PEEL
// ============================================================================

saveValue =
  preventedDeathProbability
  * protectedHeroStrategicValue
  * postSaveFightContribution
  - saveCooldownOpportunityCost
  - casterDeathRisk
  - saveOverlapPenalty;

protectedHeroStrategicValue =
  currentDpsContribution
  + remainingCooldownImpact
  + buybackStateValue
  + objectiveConversionValue;

peelValue =
  enemyThreatToBackline
  + protectedAllyValue
  + damageInterrupted
  + repositionValue
  - peelResourceCost
  - frontlineCollapseRisk;

// Reservar:
// - 1 save principal;
// - 1 save backup;
// - 1 peel primário;
// - 1 interrupt de channeling.

  + allyProtectionCoverage
  + escapeRouteQuality
  + auraCoverage
  + initiationAngle
  - enemyReachability
  - aoeExposure
  - flankThreat
  - isolationRisk;

carryPositionScore =
  attackUptime
  + peelCoverage
  + safeTargetAvailability
  + escapeRoute
  - enemyGapClose
  - enemyControlCoverage
  - flankThreat;

// ============================================================================
// 12. USO DE ULTIMATES
// ============================================================================

ultimateUseScore =
  immediateFightSwing
  + targetStrategicValue
  + targetsAffected
  + killConversion
  + allySaveValue
  + objectiveFollowUp
  + interruptValue
  - cooldownOpportunityCost
  - overkill
  - enemyDisengageProbability
  - missedUltimateRisk;

// Não usar regra fixa de "só ultar 3 alvos".
// Ultimate em 1 alvo pode ser correta se:
// - alvo é core sem buyback;
// - morte abre Roshan/high ground;
// - alvo está channeling;
// - alvo possui recurso estratégico decisivo.

// ============================================================================
// 13. TROCA DE ALVO
// ============================================================================

targetSwitchTrigger =
  originalTargetBecomesInvalid
  || originalTargetUsesImmunity
  || newHighThreatTargetAppears
  || channelMustBeInterrupted
  || originalTargetReceivesMultipleSaves
  || newTargetLosesEscape
  || committedDamageAlreadyEnough;

targetSwitchPenalty =
  lostTravelTime
  + lostAttackUptime
  + abandonedComboValue
  + communicationDelay;

Trocar apenas se:

newTargetScore >
  currentTargetScore
  + targetSwitchPenalty
  + switchThreshold;

// 31. PSEUDOCÓDIGO PRINCIPAL
// ============================================================================

export function updateCombatAi(input: {
  gameState: RawGameState;
  encounter: CombatEncounterState;
  teamId: string;
  blackboard: TeamfightBlackboard;
  deltaTime: number;
}): CombatAiOutput {
  const context = analyzeCombatContext(input);
  const encounterType = classifyEncounter(context);

  const phase = updateCombatPhase({
    previousPhase: input.blackboard.phase,
    encounterType,
    context,
  });

  const assignments = assignDynamicCombatRoles(context);

  const targetScores = scoreAllTargets({
    context,
    assignments,
    blackboard: input.blackboard,
  });

  const focusPlan = selectFocusTargets({
    targetScores,
    currentPrimaryTargetId: input.blackboard.primaryTargetId,
    context,
  });

  const controlPlan = coordinateControls({
    context,
    focusPlan,
    existingReservations: input.blackboard.controlReservations,
  });

  const damagePlan = coordinateDamage({
    context,
    focusPlan,
    existingReservations: input.blackboard.damageReservations,
  });

  const savePlan = coordinateSaves({
    context,
    existingReservations: input.blackboard.saveReservations,
  });

  const formationPlan = updateFormationPlan({
    context,
    assignments,
    focusPlan,
  });

  const playerActions = context.alliedHeroes.map((hero) => {
    const candidateActions = generateCombatActions({
      hero,
      encounterType,
      phase,
      focusPlan,
      controlPlan,
      damagePlan,
      savePlan,
      formationPlan,
      context,
    });

    const selectedAction = selectCombatAction({
      hero,
      candidateActions,
      context,
    });

    return applyCombatExecutionModel({
      hero,
      action: selectedAction,
      context,
    });
  });

  return {
    encounterType,
    phase,
    assignments,
    focusPlan,
    controlPlan,
    damagePlan,
    savePlan,
    formationPlan,
    playerActions,
  };
}

export function updateLaningFightAi(input: {
  laneContext: LaneFightContext;
  playerId: string;
  currentIntent?: LaneFightIntent;
}): LaneFightDecision {
  const candidates = [
    { intent: "short_trade", score: scoreShortTrade(input.laneContext) },
    { intent: "extended_trade", score: scoreExtendedTrade(input.laneContext) },
    { intent: "all_in", score: scoreAllIn(input.laneContext) },
    { intent: "secure_ranged_creep", score: scoreRangedCreepContest(input.laneContext) },
    { intent: "contest_pull", score: scorePullContest(input.laneContext) },
    { intent: "prepare_dive", score: scoreDive(input.laneContext) },
    { intent: "force_reposition", score: scoreLaneRetreat(input.laneContext) },
  ];

  return selectLaneIntent(candidates);
}

// ============================================================================
// 33. IA ESPECÍFICA DE SKIRMISH
// ============================================================================

export function updateSkirmishAi(input: {
  skirmishContext: SkirmishContext;
  teamId: string;
}): SkirmishDecision {
  const engageScore = scoreSkirmishEngage(input.skirmishContext);
  const reinforceScore = scoreReinforcement(input.skirmishContext);
  const disengageScore = scoreSkirmishDisengage(input.skirmishContext);

  if (disengageScore > engageScore) {
    return { decision: "disengage" };
  }

  if (reinforceScore > engageScore) {
    return { decision: "send_support" };
  }

  if (engageScore > COMBAT_AI_RULES.skirmish.commitThreshold) {
    return { decision: "full_commit" };
  }

  return { decision: "ignore" };
}
