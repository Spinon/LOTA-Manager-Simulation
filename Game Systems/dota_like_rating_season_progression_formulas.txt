================================================================================
1. TEAM RATING
================================================================================

teamRating =
  playerRatingsAverage
  + synergyRating
  + coachingRating
  + draftRating
  + metaReadiness
  - fatiguePenalty
  + moraleBonus

================================================================================
2. PLAYER RATING
================================================================================

  mechanics * 0.18
  + laning * 0.14
  + mapAwareness * 0.14
  + teamfight * 0.14
  + communication * 0.10
  + discipline * 0.10
  + heroPoolDepth * 0.10

================================================================================
3. ELO UPDATE
================================================================================

expectedScore =
  1 / (1 + 10 ^ ((opponentRating - teamRating) / 400))

ratingChange =
  kFactor * (actualScore - expectedScore)

actualScore:
  win = 1
  loss = 0

kFactor sugerido:
  16 para times estáveis
  24 para times novos
  32 para início de temporada

================================================================================
4. FORM DECAY
================================================================================

recentForm =
  weightedAverage(lastMatches)

formDecay:

currentForm =
  previousForm * decayFactor + latestPerformance * (1 - decayFactor)

decayFactor = 0.85

  1 - currentMetaAdaptation / 100

================================================================================
15. FUNÇÕES TYPESCRIPT
================================================================================

export function expectedEloScore(rating: number, opponentRating: number): number {
  return 1 / (1 + Math.pow(10, (opponentRating - rating) / 400));
}

export function eloChange(rating: number, opponentRating: number, actualScore: number, kFactor = 24): number {
  const expected = expectedEloScore(rating, opponentRating);
  return kFactor * (actualScore - expected);
}

export function bestOfThree(p: number): number {
  return p * p * (3 - 2 * p);
}

export function bestOfFive(p: number): number {
  return Math.pow(p, 3) + 3 * Math.pow(p, 3) * (1 - p) + 6 * Math.pow(p, 3) * Math.pow(1 - p, 2);
}

export function sampleSizeConfidence(n: number): number {
  return 1 - Math.exp(-n / 10);
}
