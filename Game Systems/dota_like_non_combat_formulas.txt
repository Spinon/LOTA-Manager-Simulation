// dota_like_non_combat_formulas.txt
// Fórmulas fora do combate para um simulador de manager de MOBA inspirado em Dota-like.
// Foco: economia, experiência, tempo, mapa, creeps, objetivos, draft, progressão de jogadores e simulação gerencial.
// Observação: Dota real muda por patch e possui exceções. Use este arquivo como base configurável.

================================================================================
0. PRINCÍPIO DE DESIGN
================================================================================

Separar o jogo em 3 camadas:

1. Regras de partida:
   - tempo
   - waves
   - runas
   - jungle
   - objetivos
   - economia
   - experiência
   - respawn
   - buyback

2. Simulação estratégica:
   - farm
   - rotação
   - pressão de mapa
   - visão
   - controle de objetivos
   - vantagem de ouro
   - vantagem de experiência
   - risco de morte
   - timing de item

3. Camada de manager:
   - treino
   - forma/fadiga
   - moral
   - sinergia
   - hero pool
   - draft
   - scrims
   - scouting
   - adaptação ao meta
   - performance em torneios

================================================================================
1. TEMPO DE PARTIDA
================================================================================

Representação:

gameTimeSeconds = elapsedSeconds
gameTimeMinutes = floor(gameTimeSeconds / 60)

Formato:

minutes = floor(gameTimeSeconds / 60)
seconds = gameTimeSeconds % 60

isPreGame = gameTimeSeconds < 0
isGameStarted = gameTimeSeconds >= 0

Ticks de simulação:

simulationTickSeconds = 1

ticksElapsed = floor(gameTimeSeconds / simulationTickSeconds)

Para simulação rápida:

simulationTickSeconds = 5 ou 10

================================================================================
2. OURO PASSIVO
================================================================================

Modelo Dota-like configurável:

passiveGoldPerMinute = valueByGameTime(gameTimeMinutes)

passiveGoldPerSecond = passiveGoldPerMinute / 60

goldFromPassiveThisTick =
  passiveGoldPerSecond * tickDurationSeconds

Tabela sugerida para manager:

0-4 min:    100 GPM
5-19 min:   106 GPM
20-39 min:  112 GPM
40+ min:    120 GPM

Implementação:

function passiveGpmAtMinute(minute) {
  if (minute >= 40) return 120;
  if (minute >= 20) return 112;
  if (minute >= 5) return 106;
  return 100;
}

3. OURO CONFIÁVEL E NÃO CONFIÁVEL
================================================================================

Dota-like usa duas categorias:

reliableGold
unreliableGold

Total:

totalGold = reliableGold + unreliableGold

Ganho confiável, exemplos:
- bounty rune
- alguns objetivos
- algumas fontes especiais

Ganho não confiável, exemplos:
- farm de creeps
- farm de neutrals
- passivo, se desejado para seu manager

Compra de item:

function spendGold(cost) {
  reliableSpent = min(reliableGold, cost)
  remainingCost = cost - reliableSpent
  unreliableSpent = min(unreliableGold, remainingCost)

  reliableGold -= reliableSpent
  unreliableGold -= unreliableSpent
}

Morte:

goldLossOnDeath = netWorth / 40

unreliableGoldLost = min(unreliableGold, goldLossOnDeath)
unreliableGold -= unreliableGoldLost

================================================================================
4. NET WORTH
================================================================================

Net worth do herói:

heroNetWorth =
  reliableGold
  + unreliableGold
  + sum(itemCurrentValue)
  + sum(consumedPermanentItemValue)
  + optionalStashValue

Para manager, recomendo:

heroNetWorth =
  currentGold
  + inventoryValue
  + backpackValue * backpackValueWeight
  + consumedPermanentValue

backpackValueWeight = 1.0 se o item conta integralmente
backpackValueWeight = 0.5 se quiser reduzir impacto em avaliação de força imediata

Net worth do time:

teamNetWorth = sum(heroNetWorth for hero in team)

Diferença de net worth:

netWorthDiff = teamA.netWorth - teamB.netWorth

Vantagem percentual:

netWorthLeadPct =
  netWorthDiff / max(teamA.netWorth, teamB.netWorth)

killGold =

streakGold =
  if killStreak < 3: 0
  if killStreak == 3: 60
  if killStreak == 4: 120
  if killStreak == 5: 180
  if killStreak == 6: 240
  if killStreak == 7: 300

6. ASSIST GOLD
================================================================================

Modelo Dota-like para dividir recompensa:

assistGoldTotal =
  (assistBaseGold + victimNetWorth * victimNetWorthCoefficient)
  * netWorthDifferenceFactor

assistGoldPerHero =
  assistGoldTotal / numberOfAssistingHeroes

Valores sugeridos:

assistBaseGold = 30
victimNetWorthCoefficient = 0.038

Fator de comeback:

if killingTeamNetWorth < victimTeamNetWorth:
  netWorthDifferenceFactor =
    1 + (victimTeamNetWorth - killingTeamNetWorth) / victimTeamNetWorth

else:
  netWorthDifferenceFactor =
    1 - (killingTeamNetWorth - victimTeamNetWorth) / killingTeamNetWorth

Limitar para não quebrar o manager:

netWorthDifferenceFactor =
  clamp(netWorthDifferenceFactor, 0.6, 1.8)

Função:

function getLevelFromXp(totalXp) {
  level = 1

  for each nextLevel from 2 to 30:
    if totalXp >= XP_TO_REACH_LEVEL[nextLevel]:
      level = nextLevel
    else:
      break

  return level
}

Progresso no nível:

xpIntoCurrentLevel =
  totalXp - XP_TO_REACH_LEVEL[currentLevel]

xpNeededForNextLevel =
  XP_TO_REACH_LEVEL[currentLevel + 1] - XP_TO_REACH_LEVEL[currentLevel]

levelProgress =
  xpIntoCurrentLevel / xpNeededForNextLevel

================================================================================
14. EQUILÍBRIO DE LANE
================================================================================

Lane equilibrium é a posição média da wave em relação ao mapa.

lanePosition:
  -1.0 = perto da base aliada
   0.0 = centro da lane
   1.0 = perto da base inimiga

Atualização:

lanePushPowerAllied =
  alliedCreepDps
  + alliedHeroWaveClear
  + alliedSummonPressure
  + alliedCatapultPressure

lanePushPowerEnemy =
  enemyCreepDps
  + enemyHeroWaveClear
  + enemySummonPressure
  + enemyCatapultPressure

laneDelta =
  (lanePushPowerAllied - lanePushPowerEnemy)
  / laneResistance

lanePosition =
  clamp(lanePosition + laneDelta * tickDuration, -1, 1)

Risco de gank:

gankRisk =
  baseGankRisk
  + max(0, lanePosition) * overextensionRisk
  - alliedVisionSafety
  - nearbyAlliedSupport

Farm safety:

farmSafety =
  1 - gankRisk

export function clamp(value: number, min: number, max: number): number {
  return Math.max(min, Math.min(max, value));
}

export function sigmoid(x: number): number {
  return 1 / (1 + Math.exp(-x));
}

export function passiveGpmAtMinute(minute: number): number {
  if (minute >= 40) return 120;
  if (minute >= 20) return 112;
  if (minute >= 5) return 106;
  return 100;
}

export function bountyRuneGold(gameTimeMinutes: number): number {
  return 40 + 6 * Math.floor(gameTimeMinutes / 4);
}

export function buybackCost(netWorth: number): number {
  return Math.floor(200 + netWorth / 13);
}

export function deathGoldLoss(netWorth: number): number {
  return netWorth / 40;
}

export function killGold(
  victimLevel: number,
  streakGold = 0,
  isFirstBlood = false
): number {
  return 125 + victimLevel * 8 + streakGold + (isFirstBlood ? 135 : 0);
}

export function assistGoldPerHero(
  victimNetWorth: number,
  killingTeamNetWorth: number,
  victimTeamNetWorth: number,
  numberOfAssistingHeroes: number
): number {
  const assistBaseGold = 30;
  const coefficient = 0.038;

  let factor: number;

  if (killingTeamNetWorth < victimTeamNetWorth) {
    factor = 1 + (victimTeamNetWorth - killingTeamNetWorth) / victimTeamNetWorth;
  } else {
    factor = 1 - (killingTeamNetWorth - victimTeamNetWorth) / killingTeamNetWorth;
  }

  factor = clamp(factor, 0.6, 1.8);

  return ((assistBaseGold + victimNetWorth * coefficient) * factor) / numberOfAssistingHeroes;
}

export function expectedTimeToItemSeconds(
  itemCost: number,
  currentGold: number,
  expectedGpm: number
): number {
  const goldNeeded = Math.max(0, itemCost - currentGold);
  return goldNeeded / (expectedGpm / 60);
}

export function getLevelFromXp(totalXp: number, xpTable: Record<number, number>): number {
  let level = 1;

  for (let nextLevel = 2; nextLevel <= 30; nextLevel++) {
    if (totalXp >= xpTable[nextLevel]) {
      level = nextLevel;
    } else {
      break;
    }
  }

  return level;
}

export function bestOfThreeWinProbability(singleGameWinProbability: number): number {
  const p = singleGameWinProbability;
  return p * p * (3 - 2 * p);
}

export function bestOfFiveWinProbability(singleGameWinProbability: number): number {
  const p = singleGameWinProbability;
  return Math.pow(p, 3) + 3 * Math.pow(p, 3) * (1 - p) + 6 * Math.pow(p, 3) * Math.pow(1 - p, 2);
}
